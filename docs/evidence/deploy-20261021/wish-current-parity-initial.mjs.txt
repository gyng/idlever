import {launchBrowser} from '../../tools/browser.mjs';
import {readFileSync,writeFileSync,mkdirSync} from 'node:fs';
import {gunzipSync} from 'node:zlib';
import {createHash} from 'node:crypto';
import assert from 'node:assert/strict';
const entries=JSON.parse(readFileSync('docs/evidence/cut222/wish-inputs.json')).entries;
const out='target/deploy-20261021/wish-current-parity';mkdirSync(out);
const browser=await launchBrowser({gpu:false});const records=[];
const sha=b=>createHash('sha256').update(b).digest('hex');
try {
 const page=await browser.newPage({viewport:{width:402,height:681},reducedMotion:'reduce'});
 await page.goto('https://dev.idlever.awoo.party/?dev=1');
 await page.getByRole('button',{name:'New game',exact:true}).click();
 await page.waitForFunction(()=>window.__idlever?.booted);
 assert.equal(await page.evaluate(()=>window.__idlever.kind),'wasm');
 for(const entry of entries){
  const bytes=readFileSync(entry.home);assert.equal(sha(bytes),entry.homeSha256);
  const raw=gunzipSync(bytes).toString();assert.equal(sha(raw),entry.rawSha256);
  const got=await page.evaluate(async raw=>{const a=window.__idlever;a.runnerOn=false;a.go({kind:'camp'});a.lineage=await a.engine.load(raw);const report=await a.engine.runOfflineSlice(86400,true);return {saved:await a.engine.save(),report};},raw);
  const native=readFileSync('target/deploy-20261021/wish-current-native-fast/'+entry.id+'.json','utf8');
  writeFileSync(out+'/'+entry.id+'.json',got.saved,{flag:'wx'});
  assert.equal(got.saved,native,'complete current native/WASM bytes '+entry.id);

  assert.deepEqual(saved.lineage.rng,original.lineage.rng);
  records.push({id:entry.id,sha256:sha(got.saved),bytes:Buffer.byteLength(got.saved),exactCurrentNativeParity:true,originalGoldLedgerFeatsRngPreserved:true});
 }
 writeFileSync(out+'/result.json',JSON.stringify({pass:true,source:'510bd647c137447ca19c7ffcb9664c1dd340aeec',scope:'six original earned wish-history saves, current native versus actually deployed WASM, exact24h full saves',records},null,2)+'\n');
 console.log('PASS six current native/deployed-WASM whole saves, exact original Gold/ledger/history/RNG');
