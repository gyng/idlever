import assert from 'node:assert/strict';
import { writeFile, mkdir } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import { launchBrowser } from '../../tools/browser.mjs';
import { phoneReviewViewport } from '../../web/tests/lib/viewport-cases.mjs';
const [url,out,commit,version]=process.argv.slice(2);
await mkdir(out,{recursive:true});
const b=await launchBrowser({gpu:false});
const result={url,commit,at:new Date().toISOString(),viewport:phoneReviewViewport(),captureKind:'visible browser viewport; headless Chromium, not Safari or a GPU benchmark',errors:[],failedRequests:[],httpErrors:[],wasm:[]};
try {
 const page=await b.newPage({viewport:result.viewport,reducedMotion:'reduce'});
 page.setDefaultTimeout(90000);
 page.on('pageerror',e=>result.errors.push(e.message));
 page.on('requestfailed',r=>result.failedRequests.push({url:r.url(),error:r.failure()?.errorText}));
 const pending=[];
 page.on('response',r=>{if(r.status()>=400)result.httpErrors.push({url:r.url(),status:r.status()});if(new URL(r.url()).pathname.endsWith('.wasm'))pending.push((async()=>{const data=await r.body();result.wasm.push({url:r.url(),status:r.status(),bytes:data.length,sha256:createHash('sha256').update(data).digest('hex')});})().catch(e=>result.wasm.push({url:r.url(),status:r.status(),bodyUnavailable:e.message})));});
 const metadata=await page.request.get(new URL('meta.json',url).href,{headers:{'Cache-Control':'no-cache'}});
 assert.equal(metadata.status(),200);
 result.metadata=await metadata.json(); result.metadataHeaders=metadata.headers();
 assert.equal(result.metadata.commit,commit);
 assert.equal(result.metadata.schemaVersion,1);
 if(version)assert.equal(result.metadata.version,version);
 assert.match(result.metadataHeaders['content-type'],/application\/json/);
 assert.equal(result.metadataHeaders['access-control-allow-origin'],'*');
 const preview=await page.request.get(new URL(result.metadata.previewImage.url,new URL('meta.json',url)).href);
 assert.equal(preview.status(),200); assert.match(preview.headers()['content-type'],/image\//);
 await page.goto(url+'?dev=1');
 await page.getByRole('button',{name:'New game',exact:true}).waitFor();
 await page.screenshot({path:out+'/title.png'});
 await page.getByRole('button',{name:'New game',exact:true}).click();
 await page.waitForFunction(()=>window.__idlever?.booted);
 assert.equal(await page.evaluate(()=>window.__idlever.kind),'wasm');
 await page.locator('.town-tag[data-next="house"]').click();
 await page.waitForFunction(()=>window.__idlever.lineage.town.home);
 const before=await page.evaluate(()=>({seed:window.__idlever.lineage.seed,heroes:window.__idlever.lineage.hero_slots}));
 result.stateIdentity=before;result.inputMethod='Playwright mouse';result.textScale=await page.evaluate(()=>getComputedStyle(document.documentElement).fontSize);
 const raw=await page.evaluate(()=>window.__idlever.engine.save());await writeFile(out+'/new-town.session.json',raw);result.saveSha256=createHash('sha256').update(raw).digest('hex');
 await page.waitForLoadState('networkidle');
 await page.waitForTimeout(1000);
 await page.screenshot({path:out+'/home.png'});
 await page.reload();
 await page.waitForFunction(()=>window.__idlever?.booted);
 assert.equal(await page.locator('.game-menu').count(),0);
 assert.deepEqual(await page.evaluate(()=>({seed:window.__idlever.lineage.seed,heroes:window.__idlever.lineage.hero_slots})),before);
 result.engine={kind:await page.evaluate(()=>window.__idlever.kind),heroes:before.heroes.length};
 await Promise.all(pending);
 assert(result.wasm.length>0); assert(result.wasm.every(w=>w.status===200));
 assert.deepEqual(result.errors,[]); assert.deepEqual(result.failedRequests,[]); assert.deepEqual(result.httpErrors,[]);
 result.pass=true;
} catch(e){result.pass=false;result.error=e.stack;process.exitCode=1;} finally{await writeFile(out+'/smoke.json',JSON.stringify(result,null,2)+'\n');await b.close();}
console.log(JSON.stringify({url,pass:result.pass,error:result.error,wasm:result.wasm}));
