def diff(a,b,path=''):
 out=[]
 if type(a)!=type(b):return [(path,a,b)]
 if isinstance(a,dict):
  for k in a.keys()|b.keys():
   if k not in a or k not in b:out.append((path+'/'+k,'missing' if k not in a else a[k],'missing' if k not in b else b[k]))
   else:out+=diff(a[k],b[k],path+'/'+k)
 elif isinstance(a,list):
  if len(a)!=len(b):out.append((path+'/length',len(a),len(b)))
  for n,(x,y) in enumerate(zip(a,b)):out+=diff(x,y,path+'/'+str(n))
 elif a!=b:out.append((path,a,b))
 return out
